# Dynamic texts for the guidll test panel. In a localisation string they are written [Root.<name>]; guidll has the engine evaluate them for the
# player's country, so a panel can show what a trigger or a script value computes.

# text chosen by a trigger
defined_text = {
	name = GuidllTestFlag
	text = {
		trigger = { has_country_flag = guidll_test_marked }
		localization_key = guidll_test_flag_on
	}
	default = guidll_test_flag_off
}

# a script value as text
defined_text = {
	name = GuidllTestValue
	value = value:guidll_test_value
}
